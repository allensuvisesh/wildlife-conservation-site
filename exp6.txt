// Q1 DIGITAL CLOCK

function showTime() {
    var d = new Date();

    var h = d.getHours();
    var m = d.getMinutes();
    var s = d.getSeconds();

    document.getElementById("clock").innerHTML =
        h + ":" + m + ":" + s;
}

showTime();
setInterval(showTime, 1000);


// Q2 CHANGE BACKGROUND COLOR

function changeColor() {
    var r = Math.floor(Math.random() * 255);
    var g = Math.floor(Math.random() * 255);
    var b = Math.floor(Math.random() * 255);

    document.body.style.backgroundColor =
        "rgb(" + r + "," + g + "," + b + ")";
}


// Q3 INCREASE AND DECREASE FONT SIZE

function increaseSize() {
    document.getElementById("para").style.fontSize = "28px";
}

function decreaseSize() {
    document.getElementById("para").style.fontSize = "16px";
}


// Q4 MOVE AND STOP TIGER

var tiger;
var timer;

function moveTiger() {
    tiger = document.getElementById("tiger");

    clearInterval(timer);

    timer = setInterval(function() {
        tiger.style.left =
            parseInt(tiger.style.left || 0) + 5 + "px";
    }, 20);
}

function stopTiger() {
    clearInterval(timer);
}


// Q5 ARRAY OF OBJECTS

var animals = [
    {
        name: "Tiger",
        species: "Bengal Tiger",
        year: 1973
    },
    {
        name: "Elephant",
        species: "Asian Elephant",
        year: 1992
    },
    {
        name: "Panda",
        species: "Giant Panda",
        year: 2016
    },
    {
        name: "Lion",
        species: "African Lion",
        year: 2005
    }
];

function displayAnimals() {
    var result = "";

    for (var i = 0; i < animals.length; i++) {

        result += "<h3>" + animals[i].name + "</h3>";

        result += "Species: " + animals[i].species + "<br>";

        result += "Year: " + animals[i].year + "<br>";

        if (animals[i].year < 2000) {
            result += "Category: Before 2000<br>";
        }
        else {
            result += "Category: After 2000<br>";
        }

        result += "<hr>";
    }

    document.getElementById("animals").innerHTML = result;
}
